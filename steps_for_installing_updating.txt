# In order for us to actually be able to write anything, we need to configure the access control withing netconf.
# For that, we use the following config, to be imported into the datastore (should be adjusted for production):
#```
#<nacm xmlns="urn:ietf:params:xml:ns:yang:ietf-netconf-acm">
#  <enable-nacm>true</enable-nacm>
#  <read-default>permit</read-default>
#  <write-default>deny</write-default>
#  <groups>
#    <group>
#      <name>admin</name>
#      <user-name>root</user-name>
#      <user-name>enpro</user-name>
#    </group>
#  </groups>
#  <rule-list>
#    <name>admin-full-access</name>
#    <group>admin</group>
#    <rule>
#      <name>permit-all</name>
#      <module-name>*</module-name>
#      <access-operations>*</access-operations>
#      <action>permit</action>
#    </rule>
#  </rule-list>
#</nacm>
#```
# Import as follows (with nacm_init.xml having the content above):
sudo sysrepocfg --import=/home/enpro/nacm_init.xml --module ietf-netconf-acm --datastore running


# To not lose changes on restart:
sudo sysrepocfg --copy-from running --datastore startup
