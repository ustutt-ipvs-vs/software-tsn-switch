cmake_minimum_required(VERSION 3.10)
project(cnc_application)

# 1. Global Settings
# ---------------------------------------------------------
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Dependencies finden
find_package(nlohmann_json 3.2.0 REQUIRED)

add_subdirectory(../common common_build)

# 2. The Main Library (cnc_lib)
# ---------------------------------------------------------
# All cpp files that implement the core functionality
add_library(cnc_lib STATIC
    src/NetconfSession.cpp
    src/JsonImporter.cpp
    src/NetworkManager.cpp
    src/GclXmlBuilder.cpp
)

# Header paths for the library (and all who use it)
target_include_directories(cnc_lib PUBLIC
    /usr/local/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/../common/include
)

# Definitions and compile flags
target_compile_definitions(cnc_lib PUBLIC 
    NC_ENABLED_SSH_TLS
    CNC_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
)

# Link libraries (Netconf, SSH, JSON, etc.)
target_link_libraries(cnc_lib PUBLIC
    netconf2
    yang
    ssh
    common_lib
    nlohmann_json::nlohmann_json
)

# 3. Executables (Apps & Tests)
# ---------------------------------------------------------

# A. The main application
add_executable(cnc_app src/main.cpp)
target_link_libraries(cnc_app PRIVATE cnc_lib)

# B. The Netconf Wrapper Test (Old Demo)
add_executable(wrapper_test_run src/netconfWrapperDemo.cpp)
target_link_libraries(wrapper_test_run PRIVATE cnc_lib)

# C. The XML Builder Test (New Demo)
# We only link the lib because the builder code is now inside it.
add_executable(xml_builder_test src/XmlBuilderDemo.cpp)
target_link_libraries(xml_builder_test PRIVATE cnc_lib)
