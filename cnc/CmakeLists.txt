cmake_minimum_required(VERSION 3.10)
project(cnc_application)

set(CMAKE_CXX_STANDARD 17)

find_package(nlohmann_json 3.2.0 REQUIRED)

set(SOURCES
    src/main.cpp
    src/JsonImporter.cpp
)

add_executable(cnc_app ${SOURCES})

add_compile_definitions(CNC_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")

target_include_directories(cnc_app PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include                  # cnc/include (JsonImporter.h)
    ${CMAKE_CURRENT_SOURCE_DIR}/../common/include        # common/include (various common headers)
)

target_link_libraries(cnc_app PRIVATE nlohmann_json::nlohmann_json)
